cmake_minimum_required(VERSION 3.25)
project(output_expander C ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

include(cmake/warnings.cmake)

if(CMAKE_CROSSCOMPILING)
   add_executable(output-expander
    src/main.c
    src/hal.c
    src/rs485.c
    src/irq_rx.c
    src/usart.c
    src/usart_irq.c
    src/crc16.c
    src/cobs.c
    src/protocol.c
    src/dispatcher.c
    src/frame_receiver.c
    src/freestanding.c
    src/output_state.c
    src/startup.S)

  # Prevent GCC from recognizing the loops inside our memory
  # functions and replacing them with calls to themselves.
  #
  # WHY:
  # A recursive memcpy() generated by the optimizer could
  # cause an infinite recursion and stack overflow.
  #
  # This flag applies only to freestanding.c, not the rest
  # of the firmware.
  set_source_files_properties(src/freestanding.c PROPERTIES
    COMPILE_OPTIONS "-fno-tree-loop-distribute-patterns")

  target_include_directories(output-expander PRIVATE src)
  output_expander_warnings(output-expander)
  target_compile_options(output-expander PRIVATE
    -ffreestanding -fno-builtin -ffunction-sections -fdata-sections
    -fstack-usage -msmall-data-limit=0)
  target_link_options(output-expander PRIVATE
    -nostdlib -nostartfiles
    -T${CMAKE_CURRENT_SOURCE_DIR}/src/ch32v003.ld
    -Wl,--gc-sections
    -Wl,-Map,$<TARGET_FILE_DIR:output-expander>/output-expander.map
    -Wl,--print-memory-usage)
  set_target_properties(output-expander PROPERTIES SUFFIX ".elf")

  add_custom_command(TARGET output-expander POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:output-expander>
            $<TARGET_FILE_DIR:output-expander>/output-expander.bin
    COMMAND ${CMAKE_OBJDUMP} -d -S $<TARGET_FILE:output-expander>
            > $<TARGET_FILE_DIR:output-expander>/output-expander.dis
    COMMAND ${CMAKE_SIZE} $<TARGET_FILE:output-expander>
    COMMAND ${CMAKE_COMMAND}
            -DELF=$<TARGET_FILE:output-expander>
            -DSIZE_TOOL=${CMAKE_SIZE}
            -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/check_size.cmake
    VERBATIM)
else()
  include(CTest)
  include(cmake/sanitizers.cmake)
  add_library(output_expander_core
    src/smoke.c
    src/irq_rx.c
    src/crc16.c
    src/cobs.c
    src/protocol.c
    src/dispatcher.c
    src/output_state.c
    src/frame_receiver.c)
  target_include_directories(output_expander_core PUBLIC src)
  output_expander_warnings(output_expander_core)
  output_expander_sanitizers(output_expander_core)

    # Test the portable RS-485 request routing logic.
  #
  # WHAT:
  # Compile dispatcher routing tests as a host executable
  # linked against the existing portable core library.
  #
  # WHY:
  # Address filtering must be verified independently of
  # USART, interrupts, GPIO, and the physical RS-485 bus.
  # The same dispatcher implementation is used by the
  # host tests and the CH32V003 firmware.
  add_executable(output_expander_dispatcher_test
    tests/unit/test_dispatcher.c)

  target_link_libraries(output_expander_dispatcher_test
    PRIVATE output_expander_core)

  # Keep the project's strict compiler warnings and
  # memory/undefined-behavior sanitizers enabled for
  # this test, just like the existing portable tests.
  output_expander_warnings(output_expander_dispatcher_test)
  output_expander_sanitizers(output_expander_dispatcher_test)

  add_test(
    NAME dispatcher_route
    COMMAND output_expander_dispatcher_test)

  # Match the established sanitizer configuration.
  # Leak detection is disabled for these tests, while
  # AddressSanitizer's other checks remain enabled.
  set_tests_properties(dispatcher_route PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

  add_executable(output_expander_smoke_test tests/unit/test_smoke.c)
  target_link_libraries(output_expander_smoke_test PRIVATE output_expander_core)
  output_expander_warnings(output_expander_smoke_test)
  output_expander_sanitizers(output_expander_smoke_test)
  add_test(NAME portable_smoke COMMAND output_expander_smoke_test)
  set_tests_properties(portable_smoke PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

  # Verify interrupt-to-foreground queue behavior on
  # the host without requiring CH32V003 hardware.
  add_executable(output_expander_irq_rx_test
    tests/unit/test_irq_rx.c)

  target_link_libraries(output_expander_irq_rx_test
    PRIVATE output_expander_core)

  output_expander_warnings(output_expander_irq_rx_test)
  output_expander_sanitizers(output_expander_irq_rx_test)

  add_test(
    NAME irq_rx
    COMMAND output_expander_irq_rx_test)

  set_tests_properties(irq_rx PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
  
  # Test the portable CRC implementation on the host.
  #
  # The same algorithm will later be compiled into
  # CH32V003 firmware, without any hardware dependencies.
  add_executable(output_expander_crc16_test
    tests/unit/test_crc16.c)

  target_link_libraries(output_expander_crc16_test
    PRIVATE output_expander_core)

  output_expander_warnings(output_expander_crc16_test)
  output_expander_sanitizers(output_expander_crc16_test)

  add_test(
    NAME crc16
    COMMAND output_expander_crc16_test)

  set_tests_properties(crc16 PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

  # Verify COBS framing independently of the MCU.
  #
  # Both the firmware and host use the same
  # protocol-defined maximum packet capacities.
  add_executable(output_expander_cobs_test
    tests/unit/test_cobs.c)

  target_link_libraries(output_expander_cobs_test
    PRIVATE output_expander_core)

  output_expander_warnings(output_expander_cobs_test)
  output_expander_sanitizers(output_expander_cobs_test)

  add_test(
    NAME cobs
    COMMAND output_expander_cobs_test)

  set_tests_properties(cobs PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

  # Test complete packet serialization independently
  # of CH32V003 registers and USART hardware.
  #
  # This exercises COBS and CRC together with the
  # version-one wire-format validation rules.
  add_executable(output_expander_protocol_test
    tests/unit/test_protocol.c)

  target_link_libraries(output_expander_protocol_test
    PRIVATE output_expander_core)

  output_expander_warnings(output_expander_protocol_test)
  output_expander_sanitizers(output_expander_protocol_test)

  add_test(
    NAME protocol
    COMMAND output_expander_protocol_test)

  set_tests_properties(protocol PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

  # Test the delimiter-driven receiver with the actual RX
  # event queue, COBS decoder and packet validator.
  #
  # WHY:
  # No CH32V003 registers are needed to verify framing,
  # errors, oversized frames and overflow recovery.
  add_executable(output_expander_frame_receiver_test
    tests/unit/test_frame_receiver.c)

  target_link_libraries(output_expander_frame_receiver_test
    PRIVATE output_expander_core)

  output_expander_warnings(output_expander_frame_receiver_test)
  output_expander_sanitizers(output_expander_frame_receiver_test)

  add_test(
    NAME frame_receiver
    COMMAND output_expander_frame_receiver_test)

  set_tests_properties(frame_receiver PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

  # Exercise portable output-state logic without GPIO hardware.
  #
  # WHAT:
  # Compile the output-state tests against the shared
  # portable implementation.
  #
  # WHY:
  # Exhaustively verify all 65,536 logical transitions
  # with sanitizers before connecting physical outputs.
  add_executable(output_expander_output_state_test
    tests/unit/test_output_state.c)

  target_link_libraries(output_expander_output_state_test
    PRIVATE output_expander_core)

  output_expander_warnings(output_expander_output_state_test)
  output_expander_sanitizers(output_expander_output_state_test)

  add_test(
    NAME output_state
    COMMAND output_expander_output_state_test)

  set_tests_properties(output_state PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
endif()
