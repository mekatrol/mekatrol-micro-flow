cmake_minimum_required(VERSION 3.25)
project(output_expander C ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

include(cmake/warnings.cmake)

if(CMAKE_CROSSCOMPILING)
   add_executable(output-expander
    src/main.c
    src/hal.c
    src/rs485.c
    src/irq_rx.c
    src/usart.c
    src/usart_irq.c
    src/crc16.c
    src/cobs.c
    src/protocol.c
    src/dispatcher.c
    src/frame_receiver.c
    src/freestanding.c
    src/output_driver.c
    src/output_state.c
    src/command_processor.c
    src/response_tx.c
    src/commissioning.c
    src/config_fields.c    
    src/device_uid.c
    src/comms_app.c    
    src/startup.S)

  # Prevent GCC from recognizing the loops inside our memory
  # functions and replacing them with calls to themselves.
  #
  # WHY:
  # A recursive memcpy() generated by the optimizer could
  # cause an infinite recursion and stack overflow.
  #
  # This flag applies only to freestanding.c, not the rest
  # of the firmware.
  set_source_files_properties(src/freestanding.c PROPERTIES
    COMPILE_OPTIONS "-fno-tree-loop-distribute-patterns")

  target_include_directories(output-expander PRIVATE src)
  output_expander_warnings(output-expander)
  target_compile_options(output-expander PRIVATE
    -ffreestanding -fno-builtin -ffunction-sections -fdata-sections
    -fstack-usage -msmall-data-limit=0)
  target_link_options(output-expander PRIVATE
    -nostdlib -nostartfiles
    -T${CMAKE_CURRENT_SOURCE_DIR}/src/ch32v003.ld
    -Wl,--gc-sections
    -Wl,-Map,$<TARGET_FILE_DIR:output-expander>/output-expander.map
    -Wl,--print-memory-usage)
  set_target_properties(output-expander PROPERTIES SUFFIX ".elf")

  add_custom_command(TARGET output-expander POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:output-expander>
            $<TARGET_FILE_DIR:output-expander>/output-expander.bin
    COMMAND ${CMAKE_OBJDUMP} -d -S $<TARGET_FILE:output-expander>
            > $<TARGET_FILE_DIR:output-expander>/output-expander.dis
    COMMAND ${CMAKE_SIZE} $<TARGET_FILE:output-expander>
    COMMAND ${CMAKE_COMMAND}
            -DELF=$<TARGET_FILE:output-expander>
            -DSIZE_TOOL=${CMAKE_SIZE}
            -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/check_size.cmake
    VERBATIM)
else()
  include(CTest)
  include(cmake/sanitizers.cmake)
  add_library(output_expander_core
    src/smoke.c
    src/irq_rx.c
    src/crc16.c
    src/cobs.c
    src/protocol.c
    src/dispatcher.c
    src/output_state.c
    src/frame_receiver.c)
  target_include_directories(output_expander_core PUBLIC src)
  output_expander_warnings(output_expander_core)
  output_expander_sanitizers(output_expander_core)

    # Test the portable RS-485 request routing logic.
  #
  # WHAT:
  # Compile dispatcher routing tests as a host executable
  # linked against the existing portable core library.
  #
  # WHY:
  # Address filtering must be verified independently of
  # USART, interrupts, GPIO, and the physical RS-485 bus.
  # The same dispatcher implementation is used by the
  # host tests and the CH32V003 firmware.
  add_executable(output_expander_dispatcher_test
    tests/unit/test_dispatcher.c)

  target_link_libraries(output_expander_dispatcher_test
    PRIVATE output_expander_core)

  # Keep the project's strict compiler warnings and
  # memory/undefined-behavior sanitizers enabled for
  # this test, just like the existing portable tests.
  output_expander_warnings(output_expander_dispatcher_test)
  output_expander_sanitizers(output_expander_dispatcher_test)

  add_test(
    NAME dispatcher_route
    COMMAND output_expander_dispatcher_test)

  # Match the established sanitizer configuration.
  # Leak detection is disabled for these tests, while
  # AddressSanitizer's other checks remain enabled.
  set_tests_properties(dispatcher_route PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

  add_executable(output_expander_smoke_test tests/unit/test_smoke.c)
  target_link_libraries(output_expander_smoke_test PRIVATE output_expander_core)
  output_expander_warnings(output_expander_smoke_test)
  output_expander_sanitizers(output_expander_smoke_test)
  add_test(NAME portable_smoke COMMAND output_expander_smoke_test)
  set_tests_properties(portable_smoke PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

  # Verify interrupt-to-foreground queue behavior on
  # the host without requiring CH32V003 hardware.
  add_executable(output_expander_irq_rx_test
    tests/unit/test_irq_rx.c)

  target_link_libraries(output_expander_irq_rx_test
    PRIVATE output_expander_core)

  output_expander_warnings(output_expander_irq_rx_test)
  output_expander_sanitizers(output_expander_irq_rx_test)

  add_test(
    NAME irq_rx
    COMMAND output_expander_irq_rx_test)

  set_tests_properties(irq_rx PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
  
  # Test the portable CRC implementation on the host.
  #
  # The same algorithm will later be compiled into
  # CH32V003 firmware, without any hardware dependencies.
  add_executable(output_expander_crc16_test
    tests/unit/test_crc16.c)

  target_link_libraries(output_expander_crc16_test
    PRIVATE output_expander_core)

  output_expander_warnings(output_expander_crc16_test)
  output_expander_sanitizers(output_expander_crc16_test)

  add_test(
    NAME crc16
    COMMAND output_expander_crc16_test)

  set_tests_properties(crc16 PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

  # Verify COBS framing independently of the MCU.
  #
  # Both the firmware and host use the same
  # protocol-defined maximum packet capacities.
  add_executable(output_expander_cobs_test
    tests/unit/test_cobs.c)

  target_link_libraries(output_expander_cobs_test
    PRIVATE output_expander_core)

  output_expander_warnings(output_expander_cobs_test)
  output_expander_sanitizers(output_expander_cobs_test)

  add_test(
    NAME cobs
    COMMAND output_expander_cobs_test)

  set_tests_properties(cobs PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

  # Test complete packet serialization independently
  # of CH32V003 registers and USART hardware.
  #
  # This exercises COBS and CRC together with the
  # version-one wire-format validation rules.
  add_executable(output_expander_protocol_test
    tests/unit/test_protocol.c)

  target_link_libraries(output_expander_protocol_test
    PRIVATE output_expander_core)

  output_expander_warnings(output_expander_protocol_test)
  output_expander_sanitizers(output_expander_protocol_test)

  add_test(
    NAME protocol
    COMMAND output_expander_protocol_test)

  set_tests_properties(protocol PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

  # Test the delimiter-driven receiver with the actual RX
  # event queue, COBS decoder and packet validator.
  #
  # WHY:
  # No CH32V003 registers are needed to verify framing,
  # errors, oversized frames and overflow recovery.
  add_executable(output_expander_frame_receiver_test
    tests/unit/test_frame_receiver.c)

  target_link_libraries(output_expander_frame_receiver_test
    PRIVATE output_expander_core)

  output_expander_warnings(output_expander_frame_receiver_test)
  output_expander_sanitizers(output_expander_frame_receiver_test)

  add_test(
    NAME frame_receiver
    COMMAND output_expander_frame_receiver_test)

  set_tests_properties(frame_receiver PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

  # Exercise portable output-state logic without GPIO hardware.
  #
  # WHAT:
  # Compile the output-state tests against the shared
  # portable implementation.
  #
  # WHY:
  # Exhaustively verify all 65,536 logical transitions
  # with sanitizers before connecting physical outputs.
  add_executable(output_expander_output_state_test
    tests/unit/test_output_state.c)

  target_link_libraries(output_expander_output_state_test
    PRIVATE output_expander_core)

  output_expander_warnings(output_expander_output_state_test)
  output_expander_sanitizers(output_expander_output_state_test)

  add_test(
    NAME output_state
    COMMAND output_expander_output_state_test)

  set_tests_properties(output_state PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

    # Test the production output driver using a simulated HAL.
  #
  # WHAT:
  # Compile the real output driver and portable state
  # implementation with test-provided HAL functions.
  #
  # WHY:
  # This verifies the GPIO masks and operation ordering
  # without accessing CH32V003 memory-mapped registers.
  #
  # Keep this executable separate from the portable core
  # library so the real hardware HAL is never required.
  add_executable(output_expander_output_driver_test
    tests/unit/test_output_driver.c
    src/output_driver.c
    src/output_state.c)

  target_include_directories(output_expander_output_driver_test
    PRIVATE src)

  output_expander_warnings(output_expander_output_driver_test)
  output_expander_sanitizers(output_expander_output_driver_test)

  add_test(
    NAME output_driver
    COMMAND output_expander_output_driver_test)

  set_tests_properties(output_driver PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

    # Test the foreground command processor with a simulated
  # output driver, without linking CH32V003 hardware code.
  #
  # WHAT:
  # Compile the production command processor, existing
  # dispatcher, and test-provided output-driver functions.
  #
  # WHY:
  # Verify address filtering and GPIO execution ordering
  # before allowing bus traffic to control real outputs.
  add_executable(output_expander_command_processor_test
    tests/unit/test_command_processor.c
    src/command_processor.c)

  target_include_directories(output_expander_command_processor_test
    PRIVATE src)

  target_link_libraries(output_expander_command_processor_test
    PRIVATE output_expander_core)

  output_expander_warnings(output_expander_command_processor_test)
  output_expander_sanitizers(output_expander_command_processor_test)

  add_test(
    NAME command_processor
    COMMAND output_expander_command_processor_test)

  set_tests_properties(command_processor PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

    # Test the foreground RS-485 response transmission boundary.
  #
  # WHAT:
  # Compile the production response transmitter and link it
  # against the existing portable protocol implementation.
  #
  # The test provides a simulated usart_write_response().
  #
  # WHY:
  # Verify complete CRC-protected COBS serialization without
  # accessing CH32V003 registers or driving the RS-485 bus.
  #
  # A separate executable prevents the host from needing
  # the hardware USART implementation.
  add_executable(output_expander_response_tx_test
    tests/unit/test_response_tx.c
    src/response_tx.c)

  target_include_directories(output_expander_response_tx_test
    PRIVATE src)

  target_link_libraries(output_expander_response_tx_test
    PRIVATE output_expander_core)

  # Preserve the project's strict warning policy,
  # AddressSanitizer and UndefinedBehaviorSanitizer.
  output_expander_warnings(output_expander_response_tx_test)
  output_expander_sanitizers(output_expander_response_tx_test)

  add_test(
    NAME response_tx
    COMMAND output_expander_response_tx_test)

  set_tests_properties(response_tx PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

    # Validate broadcast UID-directed commissioning requests.
  #
  # WHAT:
  # Build the production commissioning validator with
  # the existing portable dispatcher and protocol code.
  #
  # WHY:
  # UID comparison and address validation must be tested
  # before connecting commissioning to EEPROM writes.
  #
  # No CH32V003 registers or hardware drivers are needed.
  add_executable(output_expander_commissioning_test
    tests/unit/test_commissioning.c
    src/commissioning.c)

  target_include_directories(output_expander_commissioning_test
    PRIVATE src)

  target_link_libraries(output_expander_commissioning_test
    PRIVATE output_expander_core)

  # Preserve the existing strict compiler diagnostics.
  output_expander_warnings(output_expander_commissioning_test)

  # Preserve AddressSanitizer and UndefinedBehaviorSanitizer.
  output_expander_sanitizers(output_expander_commissioning_test)

  add_test(
    NAME commissioning
    COMMAND output_expander_commissioning_test)

  set_tests_properties(commissioning PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

    # Test the portable configuration-field codec.
  #
  # WHAT:
  # Compile production configuration serialization
  # and validation logic without hardware dependencies.
  #
  # WHY:
  # Validate exact protocol byte order and field sizes
  # before implementing persistent configuration.
  add_executable(output_expander_config_fields_test
    tests/unit/test_config_fields.c
    src/config_fields.c)

  target_include_directories(output_expander_config_fields_test
    PRIVATE src)

  # Preserve strict compiler diagnostics and sanitizers.
  output_expander_warnings(output_expander_config_fields_test)
  output_expander_sanitizers(output_expander_config_fields_test)

  add_test(
    NAME config_fields
    COMMAND output_expander_config_fields_test)

  set_tests_properties(config_fields PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")

    # Test the complete foreground communications application.
  #
  # WHAT:
  # Exercise real request routing, command execution,
  # UID commissioning and response decisions.
  #
  # WHY:
  # The application must be tested as a whole before
  # received RS-485 traffic is allowed to change GPIO.
  #
  # The test supplies simulated output-driver and
  # response-transmitter functions. No MCU hardware,
  # EEPROM or physical RS-485 adapter is needed.
  add_executable(output_expander_comms_app_test
    tests/unit/test_comms_app.c
    src/comms_app.c
    src/device_uid.c
    src/command_processor.c
    src/commissioning.c)

  target_include_directories(output_expander_comms_app_test
    PRIVATE src)

  target_link_libraries(output_expander_comms_app_test
    PRIVATE output_expander_core)

  # Maintain the project's strict warning policy and
  # host AddressSanitizer/UndefinedBehaviorSanitizer.
  output_expander_warnings(output_expander_comms_app_test)
  output_expander_sanitizers(output_expander_comms_app_test)

  add_test(
    NAME comms_app
    COMMAND output_expander_comms_app_test)

  set_tests_properties(comms_app PROPERTIES
    ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
endif()
